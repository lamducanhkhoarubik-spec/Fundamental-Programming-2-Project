cmake_minimum_required(VERSION 3.12)
project(EniesLobbyBattleSystem LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Enable CMake's built-in testing functionality (CTest)
enable_testing()

# Create the main executable from main.cpp and your source files
# Update the source file list here if you add any extra .cpp files later
add_executable(eniesLobby_bin main.cpp eniesLobby.cpp)

# ----------------------------------------------------------------
# AUTOMATIC TESTCASE DISCOVERY AND CONFIGURATION:
# ----------------------------------------------------------------

# Find all input_*.txt files inside the testcase/ directory
file(GLOB TEST_FILES "${CMAKE_CURRENT_SOURCE_DIR}/testcase/input_*.txt")

# Define the score allocated per testcase (e.g., 3.0 points per test)
set(DIEM_MOI_CAU "3.0")

foreach(FILE_PATH ${TEST_FILES})
    # Extract the file name from the absolute file path
    get_filename_component(FILE_NAME ${FILE_PATH} NAME)
    
    # Define a clean display name for the testcase
    set(TEST_NAME "Test_${FILE_NAME}")
    
    # 1. Register the testcase with the executable and its corresponding input argument
    add_test(NAME ${TEST_NAME} COMMAND eniesLobby_bin "${FILE_PATH}")
    
    # 2. Assign labels and environment meta-data for scoring/filtering
    set_tests_properties(${TEST_NAME} PROPERTIES
        LABELS "EniesLobby_Assignment"
        # Set environment variable to make the test score accessible within CTest
        ENVIRONMENT "TEST_SCORE=${DIEM_MOI_CAU}"
    )
endforeach()
